# LLM P3（RAG 对话页）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增 `/chat` 页面：基于本地向量检索的 RAG 对话——流式生成、`[n]` 引用与源 memo 卡联动、本地多会话历史（drift 存储、可删除）；同时吸收两期终审的遗留加固项。

**Architecture:** 复用 P1/P2 的 LLM 客户端层与向量索引。新增共享检索服务（查询向量缓存 + 最低分阈值，搜索页与对话共用）；`LlmApi.chatStream` 以 SSE 解析实现流式输出；RAG 编排器每轮"检索 → 组装编号上下文 → 流式生成 → 落库"；会话与消息存 drift 新表（schema v2→v3，不参与服务端同步）。UI 为单页 + 会话历史底部抽屉，`MemoMarkdown` 扩展 `onCiteTap` 让回答中的 `[n]` 可点。

**Tech Stack:** Flutter 3.47+ / Dart 3.13+、drift（schema v3 + build_runner）、flutter_riverpod、dio 5.7（ResponseType.stream + SSE）、http_mock_adapter。

## Global Constraints

- 依赖方向：features → providers → data（api/local/llm）；UI 不直接调 dio。
- 所有 chat 表查询必须带 `accountId`；账号删除（`purgeAccount`）连带清会话；chat 数据不参与服务端同步。

```dart
/// LLM (chat/embeddings) settings, hydrated from secure storage.
/// Mirrors [AsrNotifier]'s optimistic-default-then-load shape; the epoch
/// guards against a slow hydration load clobbering a save that raced it.
class LlmNotifier extends Notifier<LlmSettings> {
  int _epoch = 0;

  @override
  LlmSettings build() {
    _load(_epoch);
    return const LlmSettings();
  }

  Future<void> _load(int epoch) async {
    final loaded = await ref.read(llmRepositoryProvider).load();
    if (epoch != _epoch) return;
    state = loaded;
  }

  Future<void> save(LlmSettings settings) async {
    _epoch++;
    state = settings;
    await ref.read(llmRepositoryProvider).save(settings);
  }

  Future<void> clear() async {
    _epoch++;
    state = const LlmSettings();
    await ref.read(llmRepositoryProvider).clear();
  }
}
```

```dart
import 'dart:convert' as convert;

import 'package:drift/drift.dart';

import 'app_database.dart';

part 'chat_dao.g.dart';

/// One citation of an assistant answer: `[n]` marker + source memo.
class ChatSource {
  const ChatSource({required this.n, required this.uid, required this.excerpt});

  final int n;
  final String uid;
  final String excerpt;

  Map<String, dynamic> toJson() => {'n': n, 'uid': uid, 'excerpt': excerpt};
}

/// Defensive JSON codec for the sources column (AGENTS.md: parse must not
/// crash on missing/mistyped fields).
String sourcesToJson(List<ChatSource> sources) =>
    convert.jsonEncode([for (final s in sources) s.toJson()]);

List<ChatSource> sourcesFromJson(String raw) {
  try {
    final json = convert.jsonDecode(raw);
    if (json is! List) return const [];
    return [
      for (final item in json)
        if (item is Map && item['n'] is int && item['uid'] is String)
          ChatSource(
              n: item['n'] as int,
              uid: item['uid'] as String,
              excerpt: item['excerpt'] is String ? item['excerpt'] as String : ''),
    ];
  } catch (_) {
    return const [];
  }
}

extension ChatMessagesRowSources on ChatMessagesRow {
  List<ChatSource> get parsedSources => sourcesFromJson(sources);
}

/// All local-database access for memory chats. Scoped by [accountId] like
/// every other DAO; chats never sync to the server.
@DriftAccessor(tables: [ChatSessions, ChatMessages])
class ChatDao extends DatabaseAccessor<AppDatabase> with _$ChatDaoMixin {
  ChatDao(super.db);

  Stream<List<ChatSessionsRow>> watchSessions(String accountId) {
    final query = select(chatSessions)
      ..where((s) => s.accountId.equals(accountId))
      ..orderBy([(s) => OrderingTerm.desc(s.updatedAt)]);
    return query.watch();
  }

  Future<ChatSessionsRow> createSession(String accountId) async {
    final now = DateTime.now();
    final id = now.microsecondsSinceEpoch.toRadixString(36);
    await into(chatSessions).insert(ChatSessionsCompanion.insert(
        accountId: accountId, id: id, createdAt: now, updatedAt: now));
    return ChatSessionsRow(
        accountId: accountId, id: id, title: '', createdAt: now, updatedAt: now);
  }

  Future<void> renameSession(String accountId, String sessionId, String title) =>
      (update(chatSessions)
            ..where((s) =>
                s.accountId.equals(accountId) & s.id.equals(sessionId)))
          .write(ChatSessionsCompanion(title: Value(title)));

  Future<void> touchSession(String accountId, String sessionId) =>
      (update(chatSessions)
            ..where((s) =>
                s.accountId.equals(accountId) & s.id.equals(sessionId)))
          .write(ChatSessionsCompanion(updatedAt: Value(DateTime.now())));

  Future<void> appendMessage(
    String accountId,
    String sessionId, {
    required String role,
    required String content,
    List<ChatSource> sources = const [],
    required DateTime now,
  }) =>
      transaction(() async {
        final maxSeq = await (selectOnly(chatMessages)
              ..addColumns([chatMessages.seq.max()])
              ..where(chatMessages.accountId.equals(accountId) &
                  chatMessages.sessionId.equals(sessionId)))
            .getSingle();
        final seq = (maxSeq.read(chatMessages.seq.max()) ?? 0) + 1;
        await into(chatMessages).insert(ChatMessagesCompanion.insert(
          accountId: accountId,
          sessionId: sessionId,
          seq: seq,
          role: role,
          content: content,
          sources: Value(sourcesToJson(sources)),
          createdAt: now,
        ));
        // First user turn names the session (flomo-style).
        final session = await (select(chatSessions)
              ..where((s) =>
                  s.accountId.equals(accountId) & s.id.equals(sessionId)))
            .getSingle();
        if (role == 'user' && session.title.isEmpty && content.trim().isNotEmpty) {
          final title = content.trim();
          await (update(chatSessions)
                ..where((s) =>
                    s.accountId.equals(accountId) & s.id.equals(sessionId)))
              .write(ChatSessionsCompanion(
                  title: Value(title.length > 30 ? title.substring(0, 30) : title),
                  updatedAt: Value(now)));
        } else {
          await (update(chatSessions)
                ..where((s) =>
                    s.accountId.equals(accountId) & s.id.equals(sessionId)))
              .write(ChatSessionsCompanion(updatedAt: Value(now)));
        }
      });

  Stream<List<ChatMessagesRow>> watchMessages(String accountId, String sessionId) {
    final query = select(chatMessages)
      ..where((m) =>
          m.accountId.equals(accountId) & m.sessionId.equals(sessionId))
      ..orderBy([(m) => OrderingTerm.asc(m.seq)]);
    return query.watch();
  }

  /// The chronologically ordered tail of a session's messages.
  Future<List<ChatMessagesRow>> lastMessages(
      String accountId, String sessionId, int limit) async {
    final query = select(chatMessages)
      ..where((m) =>
          m.accountId.equals(accountId) & m.sessionId.equals(sessionId))
      ..orderBy([(m) => OrderingTerm.desc(m.seq)])
      ..limit(limit);
    final rows = await query.get();
    return rows.reversed.toList();
  }

  Future<void> deleteSession(String accountId, String sessionId) =>
      transaction(() async {
        await (delete(chatMessages)
              ..where((m) =>
                  m.accountId.equals(accountId) &
                  m.sessionId.equals(sessionId)))
            .go();
        await (delete(chatSessions)
              ..where((s) =>
                  s.accountId.equals(accountId) & s.id.equals(sessionId)))
            .go();
      });

  Future<void> deleteAllForAccount(String accountId) =>
      transaction(() async {
        await (delete(chatMessages)
              ..where((m) => m.accountId.equals(accountId)))
            .go();
        await (delete(chatSessions)
              ..where((s) => s.accountId.equals(accountId)))
            .go();
      });
}

import 'dart:typed_data';

import '../local/app_database.dart';
import '../local/memo_codec.dart' show MemoRecord;
import '../../core/utils/vector_math.dart';
import '../repositories/llm_repository.dart';

/// A retrieval hit: the memo record plus its cosine score.
class ScoredMemo {
  const ScoredMemo(this.record, this.score);

  final MemoRecord record;
  final double score;
}

/// Shared semantic retrieval for search and chat. Adds two P2-deferred
/// guardrails: a per-(model, query) embedding cache so repeated searches /
/// follow-up questions don't re-bill, and a minimum cosine threshold so
/// weakly-related hits stay out of both surfaces.
///
/// Transport failures propagate ([LlmApiException]); callers decide
/// degradation (search hides, chat reports).
class RetrievalService {
  RetrievalService({required AppDatabase db, required LlmRepository repository})
      : _db = db,
        _repository = repository;

  static const double minScore = 0.30;
  static const int queryCacheLimit = 50;

  final AppDatabase _db;
  final LlmRepository _repository;
  final _queryCache = <String, List<double>>{};

  Future<List<ScoredMemo>> retrieve(String accountId, String query,
      {int k = 12}) async {
    final trimmed = query.trim();
    if (trimmed.isEmpty) return const [];

    final settings = await _repository.load();
    if (!settings.embeddingsConfigured) return const [];
    final model = settings.embeddingModel.trim();

    final vectors = await _db.embeddingDao.vectorsForModel(accountId, model);
    if (vectors.isEmpty) return const [];

    final queryVector = await _cachedQueryVector(model, trimmed, settings);
    if (queryVector == null) return const [];

    final top = await topKSimilar(Float32List.fromList(queryVector),
        {for (final row in vectors) row.uid: row.bytes}, k);
    final aboveThreshold = [
      for (final scored in top)
        if (scored.score >= minScore) scored
    ];
    final records = await _db.memoDao.recordsByUids(
        accountId, [for (final s in aboveThreshold) s.uid]);
    final byUid = {for (final r in records) r.memo.uid: r};
    return [
      for (final s in aboveThreshold)
        if (byUid[s.uid] != null) ScoredMemo(byUid[s.uid]!, s.score),
    ];
  }

  Future<List<double>?> _cachedQueryVector(
      String model, String query, LlmSettings settings) async {
    final key = '$model|$query';
    final hit = _queryCache[key];
    if (hit != null) return hit;
    final vectors = await _repository.embeddings(settings, inputs: [query]);
    if (vectors.isEmpty || vectors.first.isEmpty) return null;
    if (_queryCache.length >= queryCacheLimit) {
      _queryCache.remove(_queryCache.keys.first);
    }
    _queryCache[key] = vectors.first;
    return vectors.first;
  }
}

/// SSE `data:` payload lines from a byte stream (comments and blanks
/// dropped). Streaming-decoded UTF-8, so multi-byte sequences split across
/// chunk boundaries survive.
Stream<String> extractSseDataLines(Stream<List<int>> bytes) async* {
  await for (final line
      in bytes.transform(utf8.decoder).transform(const LineSplitter())) {
    if (line.startsWith('data:')) yield line.substring(5).trim();
  }
}

/// Text deltas of an OpenAI-compatible streaming chat completion.
/// Stops at `[DONE]`; malformed payloads and non-content events are
/// skipped instead of killing the stream.
Stream<String> parseChatDeltas(Stream<List<int>> bytes) async* {
  await for (final data in extractSseDataLines(bytes)) {
    if (data == '[DONE]') return;
    if (data.isEmpty) continue;
    try {
      final json = jsonDecode(data);
      if (json is! Map) continue;
      final choices = json['choices'];
      if (choices is! List || choices.isEmpty) continue;
      final first = choices.first;
      if (first is! Map) continue;
      final delta = first['delta'];
      if (delta is Map) {
        final content = delta['content'];
        if (content is String && content.isNotEmpty) yield content;
      }
    } catch (_) {
      // Skip malformed events.
    }
  }
}
```

  /// Streaming variant of [chat]: yields assistant text deltas parsed from
  /// the OpenAI-compatible SSE response. The stream errors with
  /// [LlmApiException] on transport/HTTP failures.
  Stream<String> chatStream({
    required String model,
    required List<LlmChatMessage> messages,
    int? maxTokens,
  }) async* {
    Response<ResponseBody> res;
    try {
      res = await dio.post<ResponseBody>(
        '/chat/completions',
        data: {
          'model': model,
          'messages': messages.map((m) => m.toJson()).toList(),
          'max_tokens': ?maxTokens,
          'stream': true,
        },
        options: Options(responseType: ResponseType.stream),
      );
    } on LlmApiException {
      rethrow;
    } on DioException catch (e) {
      final inner = e.error;
      if (inner is LlmApiException) throw inner;
      throw LlmApiException(_llmMessageOf(e), cause: e, kind: _kindOf(e));
    }
    final body = res.data;
    if (body == null) return;
    yield* parseChatDeltas(body.stream);
  }

import '../api/llm_api.dart';
import '../local/app_database.dart';
import '../local/chat_dao.dart';
import '../local/memo_codec.dart' show MemoRecord;
import '../repositories/llm_repository.dart';
import 'retrieval.dart';

/// One in-flight turn: citations known up front + the answer stream. The
/// deltas stream persists the completed answer (with sources) to the
/// session when it closes.
class RagChatTurn {
  const RagChatTurn({required this.sources, required this.deltas});

  final List<ChatSource> sources;
  final Stream<String> deltas;
}

/// RAG orchestration per spec §6: every turn re-retrieves, assembles a
/// numbered-excerpt context and streams the answer. Guards: fixed top-12
/// retrieval, 400-char excerpts, only the last [historyMessages] turns of
/// history ride along.
class RagChatEngine {
  RagChatEngine({
    required this.accountId,
    required AppDatabase db,
    required LlmRepository repository,
    required RetrievalService retrieval,
  })  : _db = db,
        _repository = repository,
        _retrieval = retrieval;

  static const excerptChars = 400;
  static const historyMessages = 8;

  final String accountId;
  final AppDatabase _db;
  final LlmRepository _repository;
  final RetrievalService _retrieval;

  Future<RagChatTurn> send({
    required String sessionId,
    required String question,
    required String language,
  }) async {
    final settings = await _repository.load();
    if (!settings.isConfigured) {
      throw const LlmApiException('Chat model not configured');
    }

    // History window BEFORE this turn's user message lands.
    final history = await _db.chatDao
        .lastMessages(accountId, sessionId, historyMessages);

    // Retrieval failures are non-fatal: chat degrades to no citations.
    List<ScoredMemo> hits = const [];
    try {
      hits = await _retrieval.retrieve(accountId, question);
    } catch (_) {
      // surfaced by the empty sources
    }
    final sources = [
      for (var i = 0; i < hits.length; i++)
        ChatSource(
            n: i + 1, uid: hits[i].record.memo.uid, excerpt: _clip(hits[i].record.memo.content)),
    ];

    final messages = <LlmChatMessage>[
      LlmChatMessage(role: 'system', content: _systemPrompt(language)),
      ...history.map((m) => LlmChatMessage(role: m.role, content: m.content)),
      LlmChatMessage(role: 'user', content: _userPrompt(question, hits, language)),
    ];

    final now = DateTime.now();
    await _db.chatDao.appendMessage(accountId, sessionId,
        role: 'user', content: question, now: now);

    Stream<String> persisting() async* {
      final buffer = StringBuffer();
      try {
        await for (final chunk
            in _repository.chatStream(settings, messages: messages)) {
          buffer.write(chunk);
          yield chunk;
        }
      } catch (e) {
        // Keep whatever streamed; the UI surfaces the error.
        if (buffer.isNotEmpty) {
          await _db.chatDao.appendMessage(accountId, sessionId,
              role: 'assistant',
              content: buffer.toString(),
              sources: sources,
              now: DateTime.now());
        }
        rethrow;
      }
      await _db.chatDao.appendMessage(accountId, sessionId,
          role: 'assistant',
          content: buffer.toString(),
          sources: sources,
          now: DateTime.now());
    }

    return RagChatTurn(sources: sources, deltas: persisting());
  }

  static String _clip(String content) {
    final c = content.trim();
    return c.length <= excerptChars ? c : '${c.substring(0, excerptChars)}…';
  }

  static String _systemPrompt(String language) => language == 'en'
      ? 'You are a warm journaling assistant answering ONLY from the '
        'numbered memories quoted in the user message. Do not invent '
        'facts; if the memories cannot answer, say so plainly. Cite the '
        'memories you used as [1], [2] right after the claim. Reply in '
        'English Markdown.'
      : '你是一位温暖的日记助手，只依据用户消息中引用的编号记忆回答，不要编造事实；'
        '记忆无法回答时明确说明。引用某条记忆时在相应结论后标注 [1]、[2] 编号。'
        '用中文以 Markdown 回复。';

  static String _userPrompt(
      String question, List<ScoredMemo> hits, String language) {
    final b = StringBuffer();
    if (hits.isEmpty) {
      b.writeln(language == 'en'
          ? '(No related memories found.)'
          : '（未找到相关记忆。）');
    } else {
      b.writeln(language == 'en' ? 'Related memories:' : '相关记忆：');
      for (var i = 0; i < hits.length; i++) {
        final memo = hits[i].record.memo;
        final created = memo.createTime?.toLocal();
        final date = created == null
            ? ''
            : '${created.year}-${created.month.toString().padLeft(2, '0')}-${created.day.toString().padLeft(2, '0')}';
        final tags = memo.tags.isEmpty ? '' : ' #${memo.tags.join(' #')}';
        b.writeln('[${i + 1}] $date$tags');
        b.writeln(_clip(memo.content));
      }
    }
    b.writeln();
    b.writeln(language == 'en' ? 'Question: $question' : '问题：$question');
    return b.toString();
  }
}

import '../../providers/auth_providers.dart';
import '../../providers/llm_providers.dart';
import '../../providers/rag_chat_providers.dart';
import '../../providers/sync_providers.dart' show appDatabaseProvider;

/// `/chat`: RAG conversation over the local memory mirror. One page, a
/// sessions bottom sheet, streaming answers with tappable `[n]` citations.
class ChatPage extends ConsumerStatefulWidget {
  const ChatPage({super.key});

  @override
  ConsumerState<ChatPage> createState() => _ChatPageState();
}

class _ChatPageState extends ConsumerState<ChatPage> {
  final _input = TextEditingController();
  final _scroll = ScrollController();

  String? _sessionId;
  List<ChatMessagesRow> _messages = const [];
  StreamSubscription<ChatMessagesRow>? _messagesSub;

  bool _busy = false;
  String? _error;
  StringBuffer? _pending;
  List<ChatSource> _pendingSources = const [];
  StreamSubscription<String>? _deltaSub;

  ChatDao get _chatDao => ref.read(appDatabaseProvider).chatDao;
  String get _accountId =>
      ref.read(authProvider).activeAccount?.id ?? '';

  @override
  void initState() {
    super.initState();
    Future.microtask(() async {
      final sessions = await _chatDao.watchSessions(_accountId).first;
      if (!mounted || sessions.isEmpty) return;
      setState(() => _sessionId = sessions.first.id);
    });
  }

  @override
  void dispose() {
    _deltaSub?.cancel();
    _messagesSub?.cancel();
    _input.dispose();
    _scroll.dispose();
    super.dispose();
  }

  void _subscribeMessages(String sessionId) {
    _messagesSub?.cancel();
    _messagesSub = _chatDao.watchMessages(_accountId, sessionId).listen(
        (messages) {
      if (mounted) setState(() => _messages = messages);
    });
  }

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context)!;
    final theme = Theme.of(context);
    final configured = ref.watch(llmProvider.select((s) => s.isConfigured));
    // Re-subscribe when the active session changes (incl. first creation).
    final sessionId = _sessionId;
    if (sessionId != null && _messagesSub == null) {
      Future.microtask(() => _subscribeMessages(sessionId));
    }

    return Scaffold(
      appBar: AppBar(
        title: Text(l10n.chatTitle),
        actions: [
          IconButton(
            tooltip: l10n.chatHistory,
            icon: const Icon(Icons.history_rounded),
            onPressed: _showSessions,
          ),
        ],
      ),
      body: !configured
          ? Center(
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Icon(Icons.forum_outlined,
                      size: 52, color: theme.colorScheme.onSurfaceVariant),
                  const SizedBox(height: 12),
                  Text(l10n.chatUnconfigured,
                      textAlign: TextAlign.center,
                      style: theme.textTheme.bodyMedium?.copyWith(
                          color: theme.colorScheme.onSurfaceVariant)),
                  const SizedBox(height: 12),
                  FilledButton.tonal(
                    onPressed: () => context.push('/settings/llm'),
                    child: Text(l10n.chatGoConfig),
                  ),
                ],
              ),
            )
          : Column(
              children: [
                if (_error != null)
                  Padding(
                    padding: const EdgeInsets.fromLTRB(16, 8, 16, 0),
                    child: Text(_error!,
                        style: theme.textTheme.bodySmall
                            ?.copyWith(color: theme.colorScheme.error)),
                  ),
                Expanded(
                  child: _messages.isEmpty && _pending == null
                      ? Center(
                          child: Text(l10n.chatEmpty,
                              textAlign: TextAlign.center,
                              style: theme.textTheme.bodyMedium?.copyWith(
                                  color: theme.colorScheme.onSurfaceVariant)),
                        )
                      : ListView.builder(
                          controller: _scroll,
                          padding: const EdgeInsets.all(12),
                          itemCount: _messages.length + (_pending != null ? 1 : 0),
                          itemBuilder: (context, index) {
                            if (index < _messages.length) {
                              return _bubble(context, _messages[index], null);
                            }
                            return _pendingBubble(context);
                          },
                        ),
                ),
                SafeArea(
                  child: Padding(
                    padding: const EdgeInsets.fromLTRB(12, 4, 12, 8),
                    child: Row(
                      children: [
                        Expanded(
                          child: TextField(
                            controller: _input,
                            minLines: 1,
                            maxLines: 4,
                            textInputAction: TextInputAction.send,
                            onSubmitted: (_) => _send(),
                            decoration: InputDecoration(
                                hintText: l10n.chatInputHint,
                                isDense: true,
                                border: const OutlineInputBorder()),
                          ),
                        ),
                        const SizedBox(width: 8),
                        IconButton.filled(
                          icon: const Icon(Icons.send_rounded),
                          onPressed: _busy ? null : _send,
                        ),
                      ],
                    ),
                  ),
                ),
              ],
            ),
    );
  }

  Widget _bubble(BuildContext context, ChatMessagesRow row, List<ChatSource>? overrideSources) {
    final theme = Theme.of(context);
    final isUser = row.role == 'user';
    final sources = overrideSources ?? row.parsedSources;
    return Align(
      alignment: isUser ? Alignment.centerRight : Alignment.centerLeft,
      child: Container(
        margin: const EdgeInsets.only(bottom: 10),
        padding: const EdgeInsets.all(12),
        constraints:
            BoxConstraints(maxWidth: MediaQuery.of(context).size.width * 0.82),
        decoration: BoxDecoration(
          color: isUser
              ? theme.colorScheme.primaryContainer
              : theme.colorScheme.surfaceContainerHighest,
          borderRadius: BorderRadius.circular(12),
        ),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            if (isUser)
              Text(row.content, style: theme.textTheme.bodyMedium)
            else ...[
              MemoMarkdown(
                content: row.content,
                onCiteTap: sources.isEmpty
                    ? null
                    : (n) => _openSource(sources, n),
              ),
              if (sources.isNotEmpty) ...[
                const SizedBox(height: 8),
                _sourcesRow(context, sources),
              ],
            ],
          ],
        ),
      ),
    );
  }

  Widget _pendingBubble(BuildContext context) {
    final theme = Theme.of(context);
    return Align(
      alignment: Alignment.centerLeft,
      child: Container(
        margin: const EdgeInsets.only(bottom: 10),
        padding: const EdgeInsets.all(12),
        constraints:
            BoxConstraints(maxWidth: MediaQuery.of(context).size.width * 0.82),
        decoration: BoxDecoration(
          color: theme.colorScheme.surfaceContainerHighest,
          borderRadius: BorderRadius.circular(12),
        ),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(_pending?.toString() ?? '',
                style: theme.textTheme.bodyMedium),
            const SizedBox(height: 6),
            const SizedBox(
                width: 16,
                height: 16,
                child: CircularProgressIndicator(strokeWidth: 2)),
          ],
        ),
      ),
    );
  }

  Widget _sourcesRow(BuildContext context, List<ChatSource> sources) {
    final theme = Theme.of(context);
    return Wrap(
      spacing: 6,
      runSpacing: 6,
      children: [
        for (final source in sources)
          ActionChip(
            avatar: Text('[${source.n}]',
                style: theme.textTheme.labelSmall
                    ?.copyWith(color: theme.colorScheme.primary)),
            label: Text(
              source.excerpt.isEmpty ? source.uid : source.excerpt,
              maxLines: 1,
              overflow: TextOverflow.ellipsis,
              style: theme.textTheme.labelSmall,
            ),
            onPressed: () => context.push('/memos/detail/${source.uid.split('/').last}'),
          ),
      ],
    );
  }

  void _openSource(List<ChatSource> sources, int n) {
    final hit = sources.where((s) => s.n == n).firstOrNull;
    if (hit != null) {
      context.push('/memos/detail/${hit.uid.split('/').last}');
    }
  }

  Future<void> _send() async {
    final l10n = AppLocalizations.of(context)!;
    final question = _input.text.trim();
    if (question.isEmpty || _busy) return;
    final engine = ref.read(ragChatEngineProvider);
    if (engine == null) return;

    setState(() => _error = null);
    var sessionId = _sessionId;
    try {
      if (sessionId == null) {
        final session = await _chatDao.createSession(_accountId);
        sessionId = session.id;
        setState(() => _sessionId = sessionId);
        _subscribeMessages(sessionId);
      }
      final language = Localizations.localeOf(context).languageCode;
      final turn = await engine.send(
          sessionId: sessionId, question: question, language: language);
      _input.clear();
      setState(() {
        _busy = true;
        _pending = StringBuffer();
        _pendingSources = turn.sources;
      });
      _deltaSub?.cancel();
      _deltaSub = turn.deltas.listen(
        (chunk) {
          _pending?.write(chunk);
          if (mounted) setState(() {});
        },
        onDone: () {
          if (!mounted) return;
          setState(() {
            _busy = false;
            _pending = null;
            _pendingSources = const [];
          });
        },
        onError: (Object e) {
          if (!mounted) return;
          setState(() {
            _busy = false;
            _pending = null;
            _error = llmErrorMessage(l10n, e);
          });
        },
      );
    } on Exception catch (e) {
      if (!mounted) return;
      setState(() => _error = llmErrorMessage(l10n, e));
    }
  }

  Future<void> _showSessions() async {
    final l10n = AppLocalizations.of(context)!;
    final accountId = _accountId;
    await showModalBottomSheet<void>(
      context: context,
      builder: (sheetContext) => SafeArea(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Padding(
              padding: const EdgeInsets.fromLTRB(20, 14, 20, 6),
              child: Text(l10n.chatHistory,
                  style: Theme.of(sheetContext).textTheme.titleSmall),
            ),
            Flexible(
              child: StreamBuilder<List<ChatSessionsRow>>(
                stream: _chatDao.watchSessions(accountId),
                builder: (context, snap) {
                  final sessions = snap.data ?? const <ChatSessionsRow>[];
                  if (sessions.isEmpty) {
                    return Padding(
                      padding: const EdgeInsets.all(20),
                      child: Text(l10n.chatEmpty,
                          style: Theme.of(sheetContext).textTheme.bodySmall),
                    );
                  }
                  return ListView.builder(
                    shrinkWrap: true,
                    itemCount: sessions.length,
                    itemBuilder: (context, index) {
                      final session = sessions[index];
                      return ListTile(
                        dense: true,
                        title: Text(
                            session.title.isEmpty ? l10n.chatNewSession : session.title,
                            maxLines: 1,
                            overflow: TextOverflow.ellipsis),
                        subtitle: Text(
                            '${session.updatedAt.toLocal()}'.substring(0, 16)),
                        trailing: IconButton(
                          icon: const Icon(Icons.delete_outline, size: 20),
                          onPressed: () async {
                            final confirmed = await showDialog<bool>(
                              context: sheetContext,
                              builder: (dialogContext) => AlertDialog(
                                content: Text(l10n.chatDeleteSessionConfirm),
                                actions: [
                                  TextButton(
                                    onPressed: () =>
                                        Navigator.pop(dialogContext, false),
                                    child: Text(l10n.cancel),
                                  ),
                                  FilledButton(
                                    onPressed: () =>
                                        Navigator.pop(dialogContext, true),
                                    child: Text(l10n.confirm),
                                  ),
                                ],
                              ),
                            );
                            if (confirmed != true) return;
                            await _chatDao.deleteSession(accountId, session.id);
                            if (!mounted) return;
                            if (_sessionId == session.id) {
                              setState(() => _sessionId = null);
                              _messagesSub?.cancel();
                              _messagesSub = null;
                              setState(() => _messages = const []);
                            }
                          },
                        ),
                        onTap: () {
                          Navigator.pop(sheetContext);
                          setState(() => _sessionId = session.id);
                          _messagesSub?.cancel();
                          _messagesSub = null;
                          _subscribeMessages(session.id);
                        },
                      );
                    },
                  );
                },
              ),
            ),
            Padding(
              padding: const EdgeInsets.fromLTRB(20, 6, 20, 14),
              child: FilledButton.tonalIcon(
                icon: const Icon(Icons.add_rounded, size: 18),
                label: Text(l10n.chatNewSession),
                onPressed: () {
                  Navigator.pop(sheetContext);
                  setState(() => _sessionId = null);
                  _messagesSub?.cancel();
                  _messagesSub = null;
                  setState(() => _messages = const []);
                },
              ),
            ),
          ],
        ),
      ),
    );
  }
}
