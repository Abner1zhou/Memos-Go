```dart
import 'package:dio/dio.dart';

/// Typed error surfaced by the LLM layer so UI can show friendly messages.
class LlmApiException implements Exception {
  const LlmApiException(this.message, {this.statusCode, this.cause});

  final String message;
  final int? statusCode;
  final Object? cause;

  @override
  String toString() => 'LlmApiException($statusCode): $message';
}

/// Normalizes a user-entered OpenAI-compatible endpoint: adds https:// when
/// the scheme is missing and strips trailing slashes.
String normalizeLlmBaseUrl(String raw) {
  var url = raw.trim();
  if (url.isEmpty) return url;
  if (!url.startsWith(RegExp(r'https?://'))) url = 'https://$url';
  return url.replaceAll(RegExp(r'/+$'), '');
}

/// Returns whether [raw] fails to parse as an endpoint URL (scheme + host,
/// no spaces) for the caller to map to a localized message.
bool llmBaseUrlInvalid(String raw) {
  final url = normalizeLlmBaseUrl(raw);
  if (url.isEmpty || RegExp(r'\s').hasMatch(url)) return true;
  final uri = Uri.tryParse(url);
  return uri == null || !uri.hasScheme || uri.host.isEmpty;
}

/// Creates a [Dio] client pointed at an OpenAI-compatible endpoint base
/// (e.g. `https://api.deepseek.com/v1`). [apiKey] is sent as
/// `Authorization: Bearer` on every request.
Dio createLlmDio({required String baseUrl, required String apiKey}) {
  final dio = Dio(BaseOptions(
    baseUrl: baseUrl,
    connectTimeout: const Duration(seconds: 15),
    // Long generations can take a while on slower services.
    receiveTimeout: const Duration(minutes: 3),
    sendTimeout: const Duration(seconds: 60),
    headers: {
      'Authorization': 'Bearer $apiKey',
    },
  ));
  dio.interceptors.add(InterceptorsWrapper(
    onError: (e, handler) {
      handler.reject(
        DioException(
          requestOptions: e.requestOptions,
          error: LlmApiException(
            _llmMessageOf(e),
            statusCode: e.response?.statusCode,
            cause: e,
          ),
          type: e.type,
          response: e.response,
        ),
      );
    },
  ));
  return dio;
}

/// OpenAI-style errors use `{"error": {"message": ...}}`; some compatible
/// servers answer `{"message": ...}` instead. Both are honored. The
/// fallback always carries the HTTP status (or the underlying socket/TLS
/// error) so a bare gateway answer like an empty 404 stays diagnosable.
String _llmMessageOf(DioException e) {
  final data = e.response?.data;
  if (data is Map) {
    final error = data['error'];
    if (error is Map) {
      final msg = error['message'];
      if (msg is String && msg.isNotEmpty) return msg;
    }
    final msg = data['message'];
    if (msg is String && msg.isNotEmpty) return msg;
  }
  final detail = switch (e.type) {
    DioExceptionType.connectionTimeout ||
    DioExceptionType.sendTimeout ||
    DioExceptionType.receiveTimeout =>
      'Connection timed out',
    DioExceptionType.connectionError => 'Cannot reach server',
    DioExceptionType.unknown => e.error?.toString() ?? 'Request failed',
    _ => e.response?.statusMessage ?? e.message ?? 'Request failed',
  };
  final status = e.response?.statusCode;
  return status == null ? detail : 'HTTP $status $detail';
}

/// Extracts an [LlmApiException] from a thrown object, if any.
LlmApiException? asLlmApiError(Object error) {
  if (error is LlmApiException) return error;
  if (error is DioException) {
    final inner = error.error;
    if (inner is LlmApiException) return inner;
  }
  return null;
}

/// One chat turn; [role] is `system` / `user` / `assistant`.
class LlmChatMessage {
  const LlmChatMessage({required this.role, required this.content});

  final String role;
  final String content;

  Map<String, dynamic> toJson() => {'role': role, 'content': content};
}

/// Client for an OpenAI-compatible chat-completions + embeddings endpoint.
/// P1 uses [chat] for the review digest; [embeddings] powers the settings
/// page connection test now and semantic search in P2.
class LlmApi {
  LlmApi(this.dio);

  final Dio dio;

  /// Sends a chat completion and returns the assistant text (empty string
  /// when the service returns no content).
  Future<String> chat({
    required String model,
    required List<LlmChatMessage> messages,
    int? maxTokens,
  }) async {
    try {
      final res = await dio.post<Map<dynamic, dynamic>>(
        '/chat/completions',
        data: {
          'model': model,
          'messages': messages.map((m) => m.toJson()).toList(),
          if (maxTokens != null) 'max_tokens': maxTokens,
        },
      );
      final data = (res.data ?? const {}).cast<String, dynamic>();
      final choices = data['choices'];
      if (choices is List && choices.isNotEmpty) {
        final first = choices.first;
        if (first is Map) {
          final msg = first['message'];
          if (msg is Map) return msg['content'] as String? ?? '';
        }
      }
      return '';
    } on LlmApiException {
      rethrow;
    } on DioException catch (e) {
      final inner = e.error;
      if (inner is LlmApiException) throw inner;
      throw LlmApiException(_llmMessageOf(e), cause: e);
    }
  }

  /// Embeds [inputs] and returns one vector per input string.
  Future<List<List<double>>> embeddings({
    required String model,
    required List<String> inputs,
  }) async {
    try {
      final res = await dio.post<Map<dynamic, dynamic>>(
        '/embeddings',
        data: {
          'model': model,
          'input': inputs,
        },
      );
      final data = (res.data ?? const {}).cast<String, dynamic>();
      final list = data['data'];
      if (list is! List) return const [];
      return list
          .whereType<Map>()
          .map((item) => (item['embedding'] as List? ?? const [])
              .whereType<num>()
              .map((n) => n.toDouble())
              .toList())
          .toList();
    } on LlmApiException {
      rethrow;
    } on DioException catch (e) {
      final inner = e.error;
      if (inner is LlmApiException) throw inner;
      throw LlmApiException(_llmMessageOf(e), cause: e);
    }
  }
}
```

```dart
import 'dart:convert' as convert;

import 'package:flutter_secure_storage/flutter_secure_storage.dart';

import '../api/llm_api.dart';

/// Configured LLM (chat + embeddings) endpoint. Persists as one JSON blob in
/// secure storage because it contains the API key.
class LlmSettings {
  const LlmSettings({
    this.baseUrl = '',
    this.apiKey = '',
    this.chatModel = '',
    this.embeddingModel = '',
  });

  /// OpenAI-compatible endpoint base, e.g. `https://api.deepseek.com/v1`.
  final String baseUrl;
  final String apiKey;
  final String chatModel;

  /// Optional for P1; required once P2 (vector search) lands.
  final String embeddingModel;

  bool get isConfigured =>
      baseUrl.trim().isNotEmpty &&
      apiKey.isNotEmpty &&
      chatModel.trim().isNotEmpty;

  bool get embeddingsConfigured =>
      isConfigured && embeddingModel.trim().isNotEmpty;

  LlmSettings copyWith({
    String? baseUrl,
    String? apiKey,
    String? chatModel,
    String? embeddingModel,
  }) =>
      LlmSettings(
        baseUrl: baseUrl ?? this.baseUrl,
        apiKey: apiKey ?? this.apiKey,
        chatModel: chatModel ?? this.chatModel,
        embeddingModel: embeddingModel ?? this.embeddingModel,
      );

  Map<String, dynamic> toJson() => {
        'baseUrl': baseUrl,
        'apiKey': apiKey,
        'chatModel': chatModel,
        'embeddingModel': embeddingModel,
      };

  factory LlmSettings.fromJson(Map<String, dynamic> json) => LlmSettings(
        baseUrl: json['baseUrl'] as String? ?? '',
        apiKey: json['apiKey'] as String? ?? '',
        chatModel: json['chatModel'] as String? ?? '',
        embeddingModel: json['embeddingModel'] as String? ?? '',
      );
}

/// Persists [LlmSettings] and runs chat completions / embeddings against the
/// configured OpenAI-compatible endpoint.
class LlmRepository {
  LlmRepository({FlutterSecureStorage? storage})
      : _storage = storage ?? const FlutterSecureStorage();

  static const _storageKey = 'memos_go.llmConfig';

  final FlutterSecureStorage _storage;

  /// Loads the saved settings; corrupt or missing data yields defaults.
  Future<LlmSettings> load() async {
    final raw = await _storage.read(key: _storageKey);
    if (raw == null || raw.isEmpty) return const LlmSettings();
    try {
      final json = convert.jsonDecode(raw);
      if (json is Map) return LlmSettings.fromJson(json.cast<String, dynamic>());
    } catch (_) {
      // Fall through to defaults on corrupt JSON.
    }
    return const LlmSettings();
  }

  Future<void> save(LlmSettings settings) => _storage.write(
        key: _storageKey,
        value: convert.jsonEncode(settings.toJson()),
      );

  Future<void> clear() => _storage.delete(key: _storageKey);

  Future<String> chat(
    LlmSettings settings, {
    required List<LlmChatMessage> messages,
    int? maxTokens,
  }) {
    final dio = createLlmDio(
      baseUrl: normalizeLlmBaseUrl(settings.baseUrl),
      apiKey: settings.apiKey,
    );
    return LlmApi(dio).chat(
      model: settings.chatModel.trim(),
      messages: messages,
      maxTokens: maxTokens,
    );
  }

  Future<List<List<double>>> embeddings(
    LlmSettings settings, {
    required List<String> inputs,
  }) {
    final dio = createLlmDio(
      baseUrl: normalizeLlmBaseUrl(settings.baseUrl),
      apiKey: settings.apiKey,
    );
    return LlmApi(dio).embeddings(
      model: settings.embeddingModel.trim(),
      inputs: inputs,
    );
  }
}
```

```dart
/// Configures an OpenAI-compatible LLM endpoint (chat + embeddings) for AI
/// features like the daily review digest.
class LlmSettingsPage extends ConsumerStatefulWidget {
  const LlmSettingsPage({super.key});

  @override
  ConsumerState<LlmSettingsPage> createState() => _LlmSettingsPageState();
}

class _LlmSettingsPageState extends ConsumerState<LlmSettingsPage> {
  final _baseUrlController = TextEditingController();
  final _apiKeyController = TextEditingController();
  final _chatModelController = TextEditingController();
  final _embeddingModelController = TextEditingController();

  var _busy = false;
  String? _error;

  @override
  void initState() {
    super.initState();
    Future.microtask(() async {
      final settings = await ref.read(llmRepositoryProvider).load();
      if (!mounted) return;
      _baseUrlController.text = settings.baseUrl;
      _apiKeyController.text = settings.apiKey;
      _chatModelController.text = settings.chatModel;
      _embeddingModelController.text = settings.embeddingModel;
    });
  }

  @override
  void dispose() {
    _baseUrlController.dispose();
    _apiKeyController.dispose();
    _chatModelController.dispose();
    _embeddingModelController.dispose();
    super.dispose();
  }

  LlmSettings _currentFields() => LlmSettings(
        baseUrl: normalizeLlmBaseUrl(_baseUrlController.text.trim()),
        apiKey: _apiKeyController.text.trim(),
        chatModel: _chatModelController.text.trim(),
        embeddingModel: _embeddingModelController.text.trim(),
      );

  /// Returns the settings to persist, or null (with `_error` set) when the
  /// form is invalid. An entirely empty form is allowed and simply disables
  /// AI features.
  LlmSettings? _validate() {
    final l10n = AppLocalizations.of(context)!;
    final settings = _currentFields();
    final empty = settings.baseUrl.isEmpty &&
        settings.apiKey.isEmpty &&
        settings.chatModel.isEmpty;
    if (empty) return settings;
    if (llmBaseUrlInvalid(settings.baseUrl)) {
      setState(() => _error = l10n.llmInvalidBaseUrl);
      return null;
    }
    return settings;
  }

  /// Pings the chat endpoint with a tiny completion, then (when an embedding
  /// model is set) the embeddings endpoint, surfacing each result.
  Future<void> _test() async {
    final l10n = AppLocalizations.of(context)!;
    setState(() => _error = null);
    final settings = _validate();
    if (settings == null) return;
    if (!settings.isConfigured) return;
    setState(() => _busy = true);
    final messenger = ScaffoldMessenger.of(context);
    final repo = ref.read(llmRepositoryProvider);
    try {
      await repo.chat(
        settings,
        messages: const [LlmChatMessage(role: 'user', content: 'ping')],
        maxTokens: 8,
      );
      messenger
        ..hideCurrentSnackBar()
        ..showSnackBar(SnackBar(content: Text(l10n.llmTestOk)));
      if (settings.embeddingsConfigured) {
        await repo.embeddings(settings, inputs: const ['你好', 'hello']);
        messenger
          ..hideCurrentSnackBar()
          ..showSnackBar(SnackBar(content: Text(l10n.llmEmbeddingsTestOk)));
      }
    } on LlmApiException catch (e) {
      setState(() => _error = '${l10n.llmTestFailed}: ${e.message}');
    } catch (e) {
      setState(() => _error = '${l10n.llmTestFailed}: $e');
    } finally {
      if (mounted) setState(() => _busy = false);
    }
  }

  Future<void> _save() async {
    setState(() => _error = null);
    final settings = _validate();
    if (settings == null) return;
    await ref.read(llmProvider.notifier).save(settings);
    if (mounted && context.canPop()) context.pop();
  }

  Future<void> _clear() async {
    final l10n = AppLocalizations.of(context)!;
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        content: Text(l10n.llmClearConfirm),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: Text(l10n.cancel),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: Text(l10n.confirm),
          ),
        ],
      ),
    );
    if (confirmed != true) return;
    await ref.read(llmProvider.notifier).clear();
    if (!mounted) return;
    setState(() {
      _baseUrlController.clear();
      _apiKeyController.clear();
      _chatModelController.clear();
      _embeddingModelController.clear();
      _error = null;
    });
  }

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context)!;
    final theme = Theme.of(context);
    final configured = ref.watch(llmProvider).isConfigured;

    return Scaffold(
      appBar: AppBar(title: Text(l10n.llmSettingsTitle)),
      body: ListView(
        padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 8),
        children: [
          Text(
            l10n.llmSettingsSubtitle,
            style: theme.textTheme.bodySmall
                ?.copyWith(color: theme.colorScheme.onSurfaceVariant),
          ),
          const SizedBox(height: 12),
          TextFormField(
            controller: _baseUrlController,
            keyboardType: TextInputType.url,
            autocorrect: false,
            decoration: InputDecoration(
              labelText: l10n.llmBaseUrl,
              hintText: l10n.llmBaseUrlHint,
              prefixIcon: const Icon(Icons.dns_outlined),
            ),
          ),
          const SizedBox(height: 12),
          TextFormField(
            controller: _apiKeyController,
            autocorrect: false,
            obscureText: true,
            decoration: InputDecoration(
              labelText: l10n.llmApiKey,
              prefixIcon: const Icon(Icons.key_outlined),
            ),
          ),
          const SizedBox(height: 12),
          TextFormField(
            controller: _chatModelController,
            autocorrect: false,
            decoration: InputDecoration(
              labelText: l10n.llmChatModel,
              hintText: l10n.llmChatModelHint,
              prefixIcon: const Icon(Icons.chat_bubble_outline_rounded),
            ),
          ),
          const SizedBox(height: 12),
          TextFormField(
            controller: _embeddingModelController,
            autocorrect: false,
            decoration: InputDecoration(
              labelText: l10n.llmEmbeddingModel,
              hintText: l10n.llmEmbeddingModelHint,

  }
```

```dart
import 'dart:convert' as convert;

import 'package:shared_preferences/shared_preferences.dart';

import '../../data/models/models.dart';

/// Token guardrails: each memo is clipped and each group capped before the
/// prompt is assembled, so digest cost stays bounded on busy days.
const _maxMemosPerGroup = 20;
const _maxCharsPerMemo = 400;

/// Input bundle for the daily review digest: memos created today plus the
/// "on this day in past years" groups the review page already computed.
class ReviewDigestInput {
  const ReviewDigestInput({required this.today, required this.onThisDay});

  final List<Memo> today;

  /// Year -> same-day memos written that year.
  final Map<int, List<Memo>> onThisDay;

  /// Cheap fingerprint of the inputs; a changed fingerprint invalidates the
  /// cached digest (memos added or edited today -> offer regenerate).
  String fingerprint() {
    final parts = <String>[];
    void add(Memo m) => parts.add(
        '${m.name}:${m.updateTime?.millisecondsSinceEpoch ?? m.createTime?.millisecondsSinceEpoch ?? 0}');
    for (final m in today) {
      add(m);
    }
    for (final e in onThisDay.entries) {
      for (final m in e.value) {
        add(m);
      }
    }
    return parts.join('|');
  }
}

/// System prompt for the digest. [language] is an ISO code ('zh' / 'en').
String reviewDigestSystemPrompt(String language) => language == 'en'
    ? 'You are a warm, concise journaling assistant. Answer ONLY from the '
      'memories provided in the user message; do not invent facts. Reply in '
      'English using Markdown with two sections: "Today" (3-5 bullet key '
      'points from today\'s memos) and, when past-year memories are present, '
      '"Echoes" (connections between today and those memories).'
    : '你是一位温暖、简洁的日记助手。只依据用户消息中给出的记忆回答，不要编造事实。'
      '用中文以 Markdown 回复，包含两个小节："今日要点"（今天的笔记提炼 3-5 条要点）；'
      '若提供了往年今日的记忆，再加"记忆呼应"小节，点出今天的记录与往年记忆的关联。';

/// Renders the input memos into the user message: today's memos first, then
/// past-year groups, each memo clipped to [_maxCharsPerMemo] characters and
/// each group capped at [_maxMemosPerGroup] memos.
String buildReviewDigestUserPrompt(
  ReviewDigestInput input, {
  required DateTime now,
  String language = 'zh',
}) {
  final b = StringBuffer();
  final day =
      '${now.year}-${now.month.toString().padLeft(2, '0')}-${now.day.toString().padLeft(2, '0')}';
  if (language == 'en') {
    b.writeln('Today is $day. The memos written today:');
  } else {
    b.writeln('今天是 $day。以下是用户今天的笔记：');
  }
  var i = 0;
  for (final memo in input.today.take(_maxMemosPerGroup)) {
    b.writeln('${++i}. ${_clip(memo.content)}');
  }
  if (input.onThisDay.isNotEmpty) {
    if (language == 'en') {
      b.writeln('\nMemos written on this day in past years:');
    } else {
      b.writeln('\n往年的今天，用户写过：');
    }
    final years = input.onThisDay.keys.toList()
      ..sort((a, b) => b.compareTo(a));
    for (final year in years) {
      for (final memo in input.onThisDay[year]!.take(_maxMemosPerGroup)) {
        b.writeln('- [$year] ${_clip(memo.content)}');
      }
    }
  }
  return b.toString();
}

String _clip(String content) {
  final c = content.trim();
  return c.length <= _maxCharsPerMemo ? c : '${c.substring(0, _maxCharsPerMemo)}…';
}

/// One cached digest for a given local day.
class ReviewDigestEntry {
  const ReviewDigestEntry({
    required this.content,
    required this.fingerprint,
    required this.generatedAt,
  });

  final String content;

  /// Fingerprint of [ReviewDigestInput] at generation time.
  final String fingerprint;
  final DateTime generatedAt;

  Map<String, dynamic> toJson() => {
        'content': content,
        'fingerprint': fingerprint,
        'generatedAt': generatedAt.toIso8601String(),
      };

  factory ReviewDigestEntry.fromJson(Map<String, dynamic> json) =>
      ReviewDigestEntry(
        content: json['content'] as String? ?? '',
        fingerprint: json['fingerprint'] as String? ?? '',
        generatedAt:
            DateTime.tryParse(json['generatedAt'] as String? ?? '') ??
                DateTime.fromMillisecondsSinceEpoch(0),
      );
}

/// Daily digest cache in shared preferences, one key per local day; only the
/// newest 7 day keys are kept so the cache cannot grow unboundedly.
class ReviewDigestCache {
  static const _prefix = 'memos_go.reviewDigest.';

  Future<ReviewDigestEntry?> load(String dayKey) async {
    final prefs = await SharedPreferences.getInstance();
    final raw = prefs.getString('$_prefix$dayKey');
    if (raw == null) return null;
    try {
      final json = convert.jsonDecode(raw);
      if (json is Map) {
        return ReviewDigestEntry.fromJson(json.cast<String, dynamic>());
      }
    } catch (_) {
      // Fall through to null on corrupt JSON.
    }
    return null;
  }

  Future<void> save(String dayKey, ReviewDigestEntry entry) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(
        '$_prefix$dayKey', convert.jsonEncode(entry.toJson()));
    final keys = prefs.getKeys().where((k) => k.startsWith(_prefix)).toList()
      ..sort();
    while (keys.length > 7) {
      await prefs.remove(keys.removeAt(0));
    }
  }
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:go_router/go_router.dart';

import 'review_digest.dart';

enum _Phase { unconfigured, idle, loading, ready, error }

/// "AI 摘要" card atop the daily review page: summarizes today's memos and
/// echoes past years' same-day memos via the configured LLM. Results are
/// cached per day; a changed input fingerprint keeps the card usable via
/// "重新生成" instead of re-billing automatically.
class ReviewDigestCard extends ConsumerStatefulWidget {
  const ReviewDigestCard({
    super.key,
    required this.today,
    required this.onThisDay,
  });

  final List<Memo> today;

  /// Year -> same-day memos written that year.
  final Map<int, List<Memo>> onThisDay;

  @override
  ConsumerState<ReviewDigestCard> createState() => _ReviewDigestCardState();
}

class _ReviewDigestCardState extends ConsumerState<ReviewDigestCard> {
  _Phase _phase = _Phase.idle;
  String? _content;
  String? _error;

  ReviewDigestInput get _input =>
      ReviewDigestInput(today: widget.today, onThisDay: widget.onThisDay);

  @override
  void initState() {
    super.initState();
    Future.microtask(() async {
      final settings = await ref.read(llmRepositoryProvider).load();
      if (!mounted) return;
      if (!settings.isConfigured) {
        setState(() => _phase = _Phase.unconfigured);
        return;
      }
      final cached = await ReviewDigestCache().load(_dayKey(DateTime.now()));
      if (!mounted) return;
      if (cached != null && cached.fingerprint == _input.fingerprint()) {
        setState(() {
          _phase = _Phase.ready;
          _content = cached.content;
        });
      }
    });
  }

  static String _dayKey(DateTime now) =>
      '${now.year}-${now.month.toString().padLeft(2, '0')}-${now.day.toString().padLeft(2, '0')}';

  Future<void> _generate() async {
    final l10n = AppLocalizations.of(context)!;
    setState(() {
      _phase = _Phase.loading;
      _error = null;
    });
    final repo = ref.read(llmRepositoryProvider);
    final settings = await repo.load();
    if (!mounted) return;
    if (!settings.isConfigured) {
      setState(() => _phase = _Phase.unconfigured);
      return;
    }
    final input = _input;
    final now = DateTime.now();
    final language = Localizations.localeOf(context).languageCode;
    try {
      final content = await repo.chat(
        settings,
        messages: [
          LlmChatMessage(
              role: 'system', content: reviewDigestSystemPrompt(language)),
          LlmChatMessage(
              role: 'user',
              content: buildReviewDigestUserPrompt(input,
                  now: now, language: language)),
        ],
      );
      await ReviewDigestCache().save(
        _dayKey(now),
        ReviewDigestEntry(
          content: content,
          fingerprint: input.fingerprint(),
          generatedAt: now,
        ),
      );
      if (!mounted) return;
      setState(() {
        _phase = _Phase.ready;
        _content = content;
      });
    } catch (e) {
      if (!mounted) return;
      final api = asLlmApiError(e);
      setState(() {
        _phase = _Phase.error;
        _error = '${l10n.digestFailed}: ${api?.message ?? e}';
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context)!;
    final theme = Theme.of(context);
    if (widget.today.isEmpty && widget.onThisDay.isEmpty) {
      return const SizedBox.shrink();
    }

    final Widget body;
    switch (_phase) {
      case _Phase.unconfigured:
        body = Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              l10n.digestNotConfigured,
              style: theme.textTheme.bodySmall
                  ?.copyWith(color: theme.colorScheme.onSurfaceVariant),
            ),
            const SizedBox(height: 8),
            FilledButton.tonal(
              onPressed: () => context.push('/settings/llm'),
              child: Text(l10n.digestGoConfig),
            ),
          ],
        );
      case _Phase.idle:
        body = FilledButton.tonal(
          onPressed: _generate,
          child: Text(l10n.digestGenerate),
        );
      case _Phase.loading:
        body = const Center(
          child: Padding(
            padding: EdgeInsets.all(12),
            child: SizedBox(
                width: 20, height: 20, child: CircularProgressIndicator(strokeWidth: 2)),
          ),
        );
      case _Phase.ready:
        body = Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            MemoMarkdown(content: _content ?? ''),
            Align(
              alignment: Alignment.centerLeft,
              child: TextButton.icon(
                onPressed: _generate,
                icon: const Icon(Icons.refresh_rounded, size: 16),
                label: Text(l10n.digestRegenerate),
              ),
            ),
          ],
        );
      case _Phase.error:
        body = Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(
              _error ?? '',
              style: theme.textTheme.bodySmall
                  ?.copyWith(color: theme.colorScheme.error),
            ),
            TextButton(onPressed: _generate, child: Text(l10n.retry)),
          ],
        );
    }

    return Card(
      margin: EdgeInsets.zero,
      clipBehavior: Clip.antiAlias,
      child: Padding(
        padding: const EdgeInsets.all(14),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Row(
              children: [
                Icon(Icons.auto_awesome_rounded,
                    size: 18, color: theme.colorScheme.primary),
                const SizedBox(width: 6),
                Text(l10n.digestTitle,
                    style: theme.textTheme.titleSmall
                        ?.copyWith(fontWeight: FontWeight.w700)),
              ],
            ),
            const SizedBox(height: 8),
            body,
          ],
        ),
      ),
    );
  }
```
